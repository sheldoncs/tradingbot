                SYMBOL,

                atr,
                ctx.AtrPercentile,
                emaFast,
                emaSlow,
                ctx,

                    if (SupportResistanceEngine.ShouldBlockTrade(

                            rates1H,
                            atr,
                            tradeDecision.Momentum,
                            currentPrice))
                    {
                        return (TradeAction.DoNothing, EntryModel.Nothing);
                    }
                    var reason = "";
                    if (!_m15Service.IsTradeAllowed(SYMBOL, pullBackAction, EntryModel.Continuation, tradeDecision.Momentum, out reason))
                    {
                        LogFile.Write($"⏸️ M15 structure not ready for pullback continuation — skipping entries and waiting for better setup for {SYMBOL}. Reason: {reason}");

                    }
                    TradingEventBus.Decision(new DecisionEvent
                    {
                        Symbol = ctx.Symbol,
                        Action = pullBackAction.ToString(),
                        Category = DecisionCategory.Continuation.ToString(),
                        Reason = $"Pullback Entry(V2) {pullBackAction.ToString()}",
                        Message = $"Pullback Entry(V2) {pullBackAction.ToString()}",
                        Momentum = tradeDecision.Momentum,
                        EarlyMomentum = false,
                        EmaCompressed = false,
                        Regime = DecisionCategory.Continuation.ToString(),
                        Time = DateTime.UtcNow
                    });
